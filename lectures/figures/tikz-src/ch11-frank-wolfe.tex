% ============================================
% LEFT PANEL: Projected Gradient Descent
% ============================================
\begin{scope}[shift={(0,0)}]

  % Title
  \node[font=\normalsize\bfseries, text=stblue!80!black] at (2.5,5.2) {Projected Gradient Descent};

  % Feasible polytope
  \fill[stblue!12] (0.5,0.5) -- (4,0.3) -- (4.5,2.5) -- (3,4) -- (1,3.5) -- cycle;
  \draw[stblue, line width=0.9pt]
    (0.5,0.5) -- (4,0.3) -- (4.5,2.5) -- (3,4) -- (1,3.5) -- cycle;
  \node[font=\footnotesize, text=stblue!60!black] at (2.5,1.8) {$\mathcal{C}$};

  % Current point x^k inside feasible set
  \fill[dkbrown] (3.2,2.8) circle (3pt);
  \node[font=\footnotesize, above right] at (3.2,2.8) {$x^k$};

  % Gradient step goes outside
  \coordinate (grad_target) at (5.5,3.8);
  \draw[vecstyle, terracotta!80!black] (3.2,2.8) -- (grad_target);
  \node[font=\footnotesize, text=terracotta!70!black, right] at (grad_target) {$x^k - \alpha\nabla f$};
  \fill[terracotta!60] (grad_target) circle (3pt);

  % Projection back
  \coordinate (proj_point) at (3.8,3.55);
  \draw[vecstyle, sage!80!black, densely dashed] (grad_target) -- (proj_point);
  \fill[sage] (proj_point) circle (3.5pt);
  \node[font=\footnotesize, text=sage!70!black, left] at (3.6,3.8) {$x^{k+1} = \Pi_\mathcal{C}(\cdot)$};

  % Labels
  \node[font=\footnotesize, text=terracotta!70!black] at (4.8,2.9) {gradient step};
  \node[font=\footnotesize, text=sage!70!black] at (5.4,4.3) {project};

\end{scope}

% ============================================
% RIGHT PANEL: Frank-Wolfe
% ============================================
\begin{scope}[shift={(8,0)}]

  % Title
  \node[font=\normalsize\bfseries, text=lavender!80!black] at (2.5,5.2) {Frank-Wolfe};

  % Feasible polytope
  \fill[lavender!10] (0.5,0.5) -- (4,0.3) -- (4.5,2.5) -- (3,4) -- (1,3.5) -- cycle;
  \draw[lavender, line width=0.9pt]
    (0.5,0.5) -- (4,0.3) -- (4.5,2.5) -- (3,4) -- (1,3.5) -- cycle;
  \node[font=\footnotesize, text=lavender!60!black] at (2.5,1.8) {$\mathcal{C}$};

  % Mark vertices
  \foreach \p in {(0.5,0.5),(4,0.3),(4.5,2.5),(3,4),(1,3.5)}{
    \fill[wmgray] \p circle (2pt);
  }

  % Current point x^k
  \fill[dkbrown] (3.0,2.5) circle (3pt);
  \node[font=\footnotesize, above right] at (3.0,2.5) {$x^k$};

  % Gradient arrow (for context)
  \draw[vecstyle, wmgray!70] (3.0,2.5) -- (3.9,3.3);
  \node[font=\tiny, text=wmgray] at (3.9,3.6) {$\nabla f$};

  % Linear minimizer vertex y^k
  \fill[rwgold] (0.5,0.5) circle (3.5pt);
  \node[font=\footnotesize, text=rwgold!80!black, below left] at (0.5,0.5)
    {$y^k = \displaystyle\arg\min_{y \in \mathcal{C}} \langle \nabla f, y \rangle$};

  % Frank-Wolfe direction
  \draw[vecstyle, terracotta!80!black, densely dashed] (3.0,2.5) -- (0.5,0.5);

  % Convex combination point x^{k+1}
  \coordinate (xnew) at ($(3.0,2.5)!0.35!(0.5,0.5)$);
  \fill[sage] (xnew) circle (3.5pt);
  \node[font=\footnotesize, text=sage!70!black, above left] at (xnew)
    {$x^{k+1}$};

  % Label the step
  \node[font=\footnotesize, text=terracotta!60!black, rotate=-39] at (1.5,1.2)
    {FW direction};

\end{scope}
